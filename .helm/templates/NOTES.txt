1. Build the shell and bff Docker images:
   // Ensure that it'll use local docker image registry.
   eval $(minikube docker-env)
   npm run docker:build:shell
   npm run docker:build:bff

2. Install the chart:
   helm install target ./helm

3. Get application access:
   // Expose shell
   kubectl port-forward svc/target-target-shell 3080:3080
   // Expose bff
   kubectl port-forward svc/target-target-bff 3000:3000


   open http://localhost:3080

4. Uninstall the chart:
   helm uninstall target
